java
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

import io.restassured.http.ContentType;
import org.junit.jupiter.api.BeforeAll;
import org.junit.jupiter.api.Test;

import java.util.List;
import java.util.Map;

class PedidosApiTest {
    private static String baseUrl;
    private static String token;

    @BeforeAll
    static void configurar() {
        baseUrl = System.getenv("API_BASE_URL");
        token = System.getenv("API_TOKEN");

        if (baseUrl == null || baseUrl.isBlank()) {
            throw new IllegalStateException("Defina API_BASE_URL");
        }
        if (token == null || token.isBlank()) {
            throw new IllegalStateException("Defina API_TOKEN");
        }
    }

    private static Map<String, Object> pedidoValido() {
        return Map.of(
            "itens", List.of(Map.of("sku", "SKU-1234", "quantidade", 1)),
            "cep_entrega", "01001-000"
        );
    }

    @Test
    void criaPedidoEValidaResposta() {
        String id = given()
            .baseUri(baseUrl)
            .auth().oauth2(token)
            .contentType(ContentType.JSON)
            .body(pedidoValido())
        .when()
            .post("/pedidos")
        .then()
            .statusCode(201)
            .contentType(ContentType.JSON)
            .body("id_pedido", not(isEmptyOrNullString()))
            .body("status", is(oneOf(
                "criado", "pago", "enviado", "entregue", "cancelado"
            )))
            .body("itens", notNullValue())
            .body("subtotal", notNullValue())
            .body("frete", notNullValue())
            .body("desconto", notNullValue())
            .body("total", notNullValue())
            .body("criado_em", notNullValue())
            .extract()
            .path("id_pedido");

        // Reutiliza o identificador retornado, sem presumir sua geração.
        given()
            .baseUri(baseUrl)
            .auth().oauth2(token)
        .when()
            .get("/pedidos/{id}", id)
        .then()
            .statusCode(200)
            .contentType(ContentType.JSON)
            .body("id_pedido", equalTo(id))
            .body("status", is(oneOf(
                "criado", "pago", "enviado", "entregue", "cancelado"
            )))
            .body("itens", notNullValue())
            .body("subtotal", notNullValue())
            .body("frete", notNullValue())
            .body("desconto", notNullValue())
            .body("total", notNullValue())
            .body("criado_em", notNullValue());
    }

    @Test
    void criarPedidoSemTokenRetorna401() {
        given()
            .baseUri(baseUrl)
            .contentType(ContentType.JSON)
            .body(pedidoValido())
        .when()
            .post("/pedidos")
        .then()
            .statusCode(401);
    }

    @Test
    void consultarPedidoSemTokenRetorna401() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/pedidos/PD-12345")
        .then()
            .statusCode(401);
    }

    @Test
    void consultarPedidoInexistenteRetorna404() {
        // Substituir por ID confirmado como inexistente no ambiente de QA.
        String idInexistente = System.getenv("PEDIDO_ID_INEXISTENTE");
        if (idInexistente == null || !idInexistente.matches("PD-[0-9]{5}")) {
            throw new IllegalStateException(
                "Defina PEDIDO_ID_INEXISTENTE com ID válido e inexistente em QA"
            );
        }

        given()
            .baseUri(baseUrl)
            .auth().oauth2(token)
        .when()
            .get("/pedidos/{id}", idInexistente)
        .then()
            .statusCode(404);
    }
}
