// src/test/java/exemplo/aurora/PedidosApiTest.java
package exemplo.aurora;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import io.restassured.response.ValidatableResponse;
import org.junit.jupiter.api.BeforeAll;
import org.junit.jupiter.api.Test;

import java.util.List;
import java.util.Map;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

class PedidosApiTest {
    private static String token;

    @BeforeAll
    static void configurar() {
        String baseUrl = System.getenv("API_BASE_URL");
        token = System.getenv("API_TOKEN");

        if (baseUrl == null || baseUrl.isBlank()) {
            throw new IllegalStateException("Defina API_BASE_URL.");
        }
        if (token == null || token.isBlank()) {
            throw new IllegalStateException("Defina API_TOKEN.");
        }

        RestAssured.baseURI = baseUrl;
    }

    private static Map<String, Object> pedidoValido() {
        return Map.of(
            "itens", List.of(Map.of("sku", "SKU-0001", "quantidade", 1)),
            "cep_entrega", "12345-678"
        );
    }

    private static ValidatableResponse postPedido(Object corpo) {
        return given()
            .auth().oauth2(token)
            .contentType(ContentType.JSON)
            .body(corpo)
        .when()
            .post("/pedidos")
        .then();
    }

    @Test
    void criaPedidoComDadosValidos() {
        postPedido(pedidoValido())
            .statusCode(201)
            .contentType(ContentType.JSON)
            .body("id_pedido", not(blankOrNullString()))
            .body("status", isIn(List.of(
                "criado", "pago", "enviado", "entregue", "cancelado"
            )))
            .body("itens", notNullValue())
            .body("subtotal", instanceOf(Number.class))
            .body("frete", instanceOf(Number.class))
            .body("desconto", instanceOf(Number.class))
            .body("total", instanceOf(Number.class))
            .body("criado_em", not(blankOrNullString()));
    }

    @Test
    void rejeitaPedidoSemItens() {
        postPedido(Map.of("itens", List.of(), "cep_entrega", "12345-678"))
            .statusCode(400);
    }

    @Test
    void rejeitaQuantidadeAbaixoDoMinimo() {
        postPedido(Map.of(
            "itens", List.of(Map.of("sku", "SKU-0001", "quantidade", 0)),
            "cep_entrega", "12345-678"
        )).statusCode(400);
    }

    @Test
    void rejeitaQuantidadeAcimaDoMaximo() {
        postPedido(Map.of(
            "itens", List.of(Map.of("sku", "SKU-0001", "quantidade", 11)),
            "cep_entrega", "12345-678"
        )).statusCode(400);
    }

    @Test
    void rejeitaSkuComFormatoInvalido() {
        postPedido(Map.of(
            "itens", List.of(Map.of("sku", "SKU-12", "quantidade", 1)),
            "cep_entrega", "12345-678"
        )).statusCode(400);
    }

    @Test
    void rejeitaCepComFormatoInvalido() {
        postPedido(Map.of(
            "itens", List.of(Map.of("sku", "SKU-0001", "quantidade", 1)),
            "cep_entrega", "12345678"
        )).statusCode(400);
    }

    @Test
    void retorna401SemToken() {
        given()
            .contentType(ContentType.JSON)
            .body(pedidoValido())
        .when()
            .post("/pedidos")
        .then()
            .statusCode(401);
    }

    @Test
    void consultaPedidoExistente() {
        String id = variavelObrigatoria("API_TEST_PEDIDO_ID");

        given()
            .auth().oauth2(token)
        .when()
            .get("/pedidos/{id}", id)
        .then()
            .statusCode(200)
            .contentType(ContentType.JSON)
            .body("id_pedido", not(blankOrNullString()))
            .body("status", isIn(List.of(
                "criado", "pago", "enviado", "entregue", "cancelado"
            )))
            .body("itens", notNullValue())
            .body("subtotal", instanceOf(Number.class))
            .body("frete", instanceOf(Number.class))
            .body("desconto", instanceOf(Number.class))
            .body("total", instanceOf(Number.class))
            .body("criado_em", not(blankOrNullString()));
    }

    @Test
    void retorna404ParaPedidoInexistente() {
        given()
            .auth().oauth2(token)
        .when()
            .get("/pedidos/{id}", "PD-99999")
        .then()
            .statusCode(404);
    }

    @Test
    void retorna401NoGetSemToken() {
        given()
        .when()
            .get("/pedidos/{id}", "PD-12345")
        .then()
            .statusCode(401);
    }

    private static String variavelObrigatoria(String nome) {
        String valor = System.getenv(nome);
        if (valor == null || valor.isBlank()) {
            throw new IllegalStateException("Defina " + nome + ".");
        }
        return valor;
    }
}
