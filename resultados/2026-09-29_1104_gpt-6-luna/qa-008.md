java
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.notNullValue;
import static org.junit.jupiter.api.Assertions.assertEquals;

import io.restassured.http.ContentType;
import io.restassured.response.Response;
import java.util.Map;
import org.junit.jupiter.api.Test;

class CancelamentoPedidoApiTest {
    private static final String BASE_URL = required("API_BASE_URL");
    private static final String TOKEN_A = required("TOKEN_CLIENTE_A");
    private static final String TOKEN_B = required("TOKEN_CLIENTE_B");

    private static String required(String name) {
        String value = System.getenv(name);
        if (value == null || value.isBlank()) {
            throw new IllegalStateException("Configure a variável de ambiente " + name);
        }
        return value;
    }

    private static String pedido(String envName) {
        return required(envName);
    }

    private static Response cancelar(String token, String id, Object body) {
        return given()
            .baseUri(BASE_URL)
            .auth().oauth2(token)
            .contentType(ContentType.JSON)
            .body(body)
        .when()
            .post("/pedidos/{id}/cancelamento", id);
    }

    private static void verificarCancelado(Response response, String id) {
        response.then()
            .statusCode(200)
            .contentType(ContentType.JSON)
            .body("id_pedido", equalTo(id))
            .body("status", equalTo("cancelado"));
    }

    @Test
    void cancelaPedidoProprioCriado() {
        String id = pedido("PEDIDO_CRIADO");
        verificarCancelado(
            cancelar(TOKEN_A, id, Map.of("motivo", "desisti da compra")),
            id
        );
    }

    @Test
    void cancelaPedidoProprioPago() {
        String id = pedido("PEDIDO_PAGO");
        verificarCancelado(
            cancelar(TOKEN_A, id, Map.of("motivo", "não quero mais")),
            id
        );
        // A solicitação de estorno é regra da HU-102; este teste não observa a integração HU-095.
    }

    @Test
    void pedidoEnviadoNaoPodeSerCancelado() {
        Response response = cancelar(
            TOKEN_A, pedido("PEDIDO_ENVIADO"), Map.of("motivo", "motivo válido")
        );
        response.then().statusCode(409);
    }

    @Test
    void pedidoEntregueNaoPodeSerCancelado() {
        cancelar(TOKEN_A, pedido("PEDIDO_ENTREGUE"), Map.of("motivo", "motivo válido"))
            .then().statusCode(409);
    }

    @Test
    void pedidoJaCanceladoNaoPodeSerCanceladoNovamente() {
        cancelar(TOKEN_A, pedido("PEDIDO_CANCELADO"), Map.of("motivo", "motivo válido"))
            .then().statusCode(409);
    }

    @Test
    void clienteNaoCancelaPedidoDeOutroCliente() {
        cancelar(TOKEN_A, pedido("PEDIDO_OUTRO_CLIENTE"), Map.of("motivo", "motivo válido"))
            .then().statusCode(404);
        // A HU também exige que o pedido não mude; verificar isso requer leitura do estado.
    }

    @Test
    void pedidoInexistenteRetorna404() {
        cancelar(TOKEN_A, pedido("PEDIDO_INEXISTENTE"), Map.of("motivo", "motivo válido"))
            .then().statusCode(404);
    }

    @Test
    void semMotivoRetorna400() {
        cancelar(TOKEN_A, pedido("PEDIDO_CRIADO"), Map.of())
            .then().statusCode(400);
    }

    @Test
    void motivoComQuatroCaracteresRetorna400() {
        cancelar(TOKEN_A, pedido("PEDIDO_CRIADO"), Map.of("motivo", "abcd"))
            .then().statusCode(400);
    }

    @Test
    void motivoComCincoCaracteresEValido() {
        String id = pedido("PEDIDO_CRIADO");
        verificarCancelado(cancelar(TOKEN_A, id, Map.of("motivo", "abcde")), id);
    }

    @Test
    void motivoComDuzentosCaracteresEValido() {
        String id = pedido("PEDIDO_CRIADO");
        String motivo = "x".repeat(200);
        verificarCancelado(cancelar(TOKEN_A, id, Map.of("motivo", motivo)), id);
    }

    @Test
    void motivoComDuzentosEUmCaracterRetorna400() {
        cancelar(TOKEN_A, pedido("PEDIDO_CRIADO"), Map.of("motivo", "x".repeat(201)))
            .then().statusCode(400);
    }

    @Test
    void semTokenRetorna401() {
        given()
            .baseUri(BASE_URL)
            .contentType(ContentType.JSON)
            .body(Map.of("motivo", "motivo válido"))
        .when()
            .post("/pedidos/{id}/cancelamento", pedido("PEDIDO_CRIADO"))
        .then()
            .statusCode(401);
    }

    @Test
    void formatoDeRespostaDeSucessoIncluiCamposDoPedido() {
        String id = pedido("PEDIDO_CRIADO");
        Response response = cancelar(TOKEN_A, id, Map.of("motivo", "desisti da compra"));
        response.then()
            .statusCode(200)
            .body("id_pedido", notNullValue())
            .body("status", equalTo("cancelado"))
            .body("itens", notNullValue());
        assertEquals(id, response.jsonPath().getString("id_pedido"));
    }
}
